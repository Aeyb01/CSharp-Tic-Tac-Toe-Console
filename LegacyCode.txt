using System;


namespace TicTacToe
{
    public class UpdateBoard
    {
        public (int row, int col) BoardInput(GameMode gameMode, TurnWinner CurrentPlayer)
        {
            string PlayerInput = UserInput(gameMode, CurrentPlayer);

            try
            {
                PlayerInput = ValidateInput(PlayerInput);
            }
            catch(ArgumentException ex)
            {
                Console.WriteLine(ex.Message);
                throw;
            }
            
            (int row, int col) = PlaceMove(PlayerInput);

            Console.WriteLine($"Coordination: {row}, {col}");

            return (row, col);
        }

        string UserInput(GameMode gameMode, TurnWinner CurrentPlayer)
        {       
            string PlayerInput;

            if (gameMode == GameMode.P1)
            {
                Console.Write("Your Turn: ");
                PlayerInput = Console.ReadLine() ?? "";
            }
            else
            {
                if (CurrentPlayer == TurnWinner.Player1)
                {
                    Console.Write("Player 1 turn: ");
                    PlayerInput = Console.ReadLine() ?? "";
                }
                else
                {
                    Console.Write("Player 2' turn: ");
                    PlayerInput = Console.ReadLine() ?? "";
                }
            }
            return PlayerInput;
        }









        string ValidateInput(string Input)
        {
            if (string.IsNullOrWhiteSpace(Input))
            {
                throw new ArgumentException("Invalid input.");
            }

            Input = Input.ToLower();

            if (Input == "exit" || Input == "a1"|| Input == "a2"|| Input == "a3"|| 
            Input == "b1"|| Input == "b2"|| Input == "b3"||
            Input == "c1"|| Input == "c2"|| Input == "c3")
            {
                return Input;
            }

            else
            {
                throw new ArgumentException("Invalid input.");
            }
        }






        (int row, int col) PlaceMove(string Input)
        {
            // Dans cette méthode on savoir quel indice l'utilisateur a saisi
            // Puis on l'envoit vers `TranslateIndex` pour remplacer cette indice
            // spécifique par un caractère de type X ou un O

            if (Input[0] == 'a')
            {
                if (Input[1] == '1')
                {
                    return (0, 0);
                }
                else if(Input[1] == '2')
                {   
                    return (0, 1);
                }
                else
                {
                    return (0, 2);
                }
            }
            else if(Input[0] == 'b')
            {
                if (Input[1] == '1')
                {
                    return (1, 0);
                }
                else if(Input[1] == '2')
                {   
                    return (1, 1);
                }
                else
                {
                    return (1, 2);
                }
            }
            else
            {
                if (Input[1] == '1')
                {
                    return (2, 0);
                }
                else if(Input[1] == '2')
                {   
                    return (2, 1);
                }
                else
                {
                    return (2, 2);
                }
            }
        }
    }
}

public static Players ReturnWinner(GameMode gameMode, GameState Condition)
        {
            // So first of all in order to know who actually won
            // We need to know the game mode in order to decide whether
            // It's going to be between an AI or a Player

            if (gameMode == GameMode.OfflineMode)
            {
                // Now we know that if the winner is X
                // that means it's Player1 else it's AI
                // Great I'll impliment it now
                if (Condition == GameState.XWins)
                {
                    // Here we return the winner which is Player
                    return Players.Player1;
                }
                else
                {
                    return Players.AI;
                }
            }
            // the same exact logic for the PVP mode
            else
            {
                if (Condition == GameState.XWins)
                {
                    return Players.Player1;
                }
                else
                {
                    return Players.Player2;
                }
            }
